# codigo de rover 1.1v
from ideaboard import IdeaBoard
import time
import board
import adafruit_hcsr04
from time import sleep

ib = IdeaBoard()

sonar = adafruit_hcsr04.HCSR04(trigger_pin=board.IO19, echo_pin=board.IO18)

ib.brightness = 0.2
BLUE = (0, 0, 255)
GREEN = (0, 255, 0)
RED = (255, 0, 0)
PURPLE = (255, 0, 255)
YELLOW = (255, 255, 0)

ib.pixel = BLUE
sleep(1)

while True:
    
    try:
        # Read distance once per loop to avoid duplicate timeout errors
        dist = sonar.distance
        print((dist,))
        
        if dist < 15.0:
            print("alto")
            ib.pixel = RED
            ib.motor_1.throttle = 0
            ib.motor_2.throttle = 0
            sleep(0.5)  # Shorter sleep keeps the robot responsive
        else:
            print("avanzando")
            ib.pixel = GREEN
            ib.motor_1.throttle = 1.0
            ib.motor_2.throttle = 1.0
            sleep(0.5)
            
    except RuntimeError:
        # Ultrasonic sensor failed to get an echo; pause briefly and retry
        print("error de lectura")
        ib.pixel = GREEN
        ib.motor_1.throttle = 1.0
        ib.motor_2.throttle = 1.0
        sleep(0.5)
