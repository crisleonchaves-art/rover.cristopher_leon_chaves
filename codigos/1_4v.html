# codigo de rover 1.4v
# Mejoras: movimiento no bloqueante, menos publicaciones, reconexion automatica

import ssl
import time

import board
import socketpool
import wifi
import adafruit_hcsr04
import adafruit_minimqtt.adafruit_minimqtt as MQTT
from adafruit_io.adafruit_io import IO_MQTT
from ideaboard import IdeaBoard

# ---------------------------------------------------------------
# CONFIGURACION
# ---------------------------------------------------------------
VELOCIDAD = 0.5            # velocidad de los motores (0 a 1)
DISTANCIA_MIN = 7.0        # cm: si hay algo mas cerca, esquiva
INTERVALO_LOOP = 0.05      # s: pausa del ciclo principal
INTERVALO_DISTANCIA = 5.0  # s: cada cuanto se publica la distancia

# ---------------------------------------------------------------
# HARDWARE
# ---------------------------------------------------------------
ib = IdeaBoard()
ib.brightness = 0.2

entrada = ib.DigitalIn(board.IO32)  # sensor de linea (False = blanco / dentro)
sonar = adafruit_hcsr04.HCSR04(trigger_pin=board.IO18, echo_pin=board.IO19)

BLUE = (0, 0, 255)      # izquierda / derecha
GREEN = (0, 255, 0)     # adelante
RED = (255, 0, 0)       # stop
YELLOW = (255, 255, 0)  # atras

# ---------------------------------------------------------------
# MOVIMIENTO
# ---------------------------------------------------------------
def mover(motor1, motor2, color, nombre):
    ib.motor_1.throttle = motor1
    ib.motor_2.throttle = motor2
    ib.pixel = color
    print(nombre)


def stop():
    mover(0, 0, RED, "stop")


def forward():
    mover(VELOCIDAD, VELOCIDAD, GREEN, "forward")


def back():
    mover(-VELOCIDAD, -VELOCIDAD, YELLOW, "back")


def left():
    mover(0, VELOCIDAD, BLUE, "left")


def right():
    mover(VELOCIDAD, 0, BLUE, "right")


# Maniobras: lista de (accion, duracion en segundos)
MANIOBRA_OBSTACULO = [(back, 0.3), (left, 2.0)]
MANIOBRA_LINEA = [(back, 0.5), (right, 0.5)]

# Estado de la maniobra en curso
maniobra = []
paso_fin = 0.0


def iniciar_maniobra(pasos):
    global maniobra
    maniobra = list(pasos)
    siguiente_paso(time.monotonic())


def siguiente_paso(ahora):
    """Ejecuta el siguiente paso de la maniobra (si queda alguno)."""
    global paso_fin
    if maniobra:
        accion, duracion = maniobra.pop(0)
        accion()
        paso_fin = ahora + duracion


def cancelar_maniobra():
    maniobra.clear()


# ---------------------------------------------------------------
# WIFI
# ---------------------------------------------------------------
try:
    from secrets import secrets
except ImportError:
    print("WiFi secrets are kept in secrets.py, please add them there!")
    raise

print("Connecting to %s" % secrets["ssid"])
wifi.radio.connect(secrets["ssid"], secrets["password"])
print("Connected to %s!" % secrets["ssid"])

# ---------------------------------------------------------------
# ADAFRUIT IO (MQTT)
# ---------------------------------------------------------------
activo = False  # si el sistema esta encendido


def connected(client):
    print("Connected to Adafruit IO!")
    # Solo nos suscribimos al feed que recibimos; line/distance/estado
    # son feeds donde solo publicamos.
    client.subscribe("off_on")


def disconnected(client):
    print("Disconnected from Adafruit IO!")


def subscribe(client, userdata, topic, granted_qos):
    print("Subscribed to {0} with QOS level {1}".format(topic, granted_qos))


def message(client, feed_id, payload):
    global activo
    print(feed_id, payload)
    if feed_id != "off_on":
        return
    comando = payload.strip().upper()
    if comando == "ON":
        activo = True
        io.publish("estado", "ACTIVADO")
        print("SISTEMA ENCENDIDO")
    elif comando == "OFF":
        activo = False
        cancelar_maniobra()
        stop()
        io.publish("estado", "DESACTIVADO")
        print("SISTEMA APAGADO")


pool = socketpool.SocketPool(wifi.radio)

mqtt_client = MQTT.MQTT(
    broker="io.adafruit.com",
    port=1883,
    username=secrets["aio_username"],
    password=secrets["aio_key"],
    socket_pool=pool,
    ssl_context=ssl.create_default_context(),
)

io = IO_MQTT(mqtt_client)
io.on_connect = connected
io.on_disconnect = disconnected
io.on_subscribe = subscribe
io.on_message = message

print("Connecting to Adafruit IO...")
io.connect()


def reconectar():
    """Detiene el rover e intenta recuperar WiFi y MQTT."""
    global activo
    activo = False
    cancelar_maniobra()
    stop()
    try:
        if not wifi.radio.connected:
            wifi.radio.connect(secrets["ssid"], secrets["password"])
        io.reconnect()
        print("Reconectado")
    except Exception as e:  # noqa: BLE001
        print("Fallo la reconexion:", e)
        time.sleep(2)


# ---------------------------------------------------------------
# PUBLICACION CONTROLADA (evita pasar el limite de Adafruit IO)
# ---------------------------------------------------------------
ultima_linea = None
ultima_pub_distancia = 0.0


def publicar_linea(valor):
    """Publica solo cuando el valor cambia."""
    global ultima_linea
    if valor != ultima_linea:
        io.publish("line", valor)
        ultima_linea = valor


def publicar_distancia(valor, ahora):
    """Publica la distancia como maximo cada INTERVALO_DISTANCIA segundos."""
    global ultima_pub_distancia
    if ahora - ultima_pub_distancia >= INTERVALO_DISTANCIA:
        io.publish("distance", valor)
        ultima_pub_distancia = ahora


# ---------------------------------------------------------------
# SENSOR DE DISTANCIA
# ---------------------------------------------------------------
ultima_distancia = 100.0  # valor inicial "seguro"


def leer_distancia():
    """Devuelve la distancia; si falla la lectura usa la ultima valida."""
    global ultima_distancia
    try:
        ultima_distancia = sonar.distance
    except RuntimeError:
        print("error de lectura")
    return ultima_distancia


# ---------------------------------------------------------------
# BUCLE PRINCIPAL
# ---------------------------------------------------------------
stop()

while True:
    ahora = time.monotonic()

    # Mantener la conexion; si falla, reconectar
    try:
        io.loop()
    except (MQTT.MMQTTException, OSError, RuntimeError) as e:
        print("Error de conexion:", e)
        reconectar()
        continue

    if not activo:
        time.sleep(INTERVALO_LOOP)
        continue

    try:
        # 1) Hay una maniobra en curso: esperar a que termine el paso
        if maniobra or ahora < paso_fin:
            if ahora >= paso_fin:
                siguiente_paso(ahora)
            time.sleep(INTERVALO_LOOP)
            continue

        # 2) Decidir segun los sensores
        if entrada.value is False:  # dentro del area (blanco)
            publicar_linea("blanco")
            distancia = leer_distancia()
            publicar_distancia(distancia, ahora)

            if distancia < DISTANCIA_MIN:
                print("alto: obstaculo")
                iniciar_maniobra(MANIOBRA_OBSTACULO)
            else:
                forward()  # se repite cada ciclo, sin bloquear
        else:  # fuera del area (negro)
            print("fuera del lugar")
            publicar_linea("negro")
            iniciar_maniobra(MANIOBRA_LINEA)

    except (MQTT.MMQTTException, OSError) as e:
        print("Error al publicar:", e)
        reconectar()
        continue

    time.sleep(INTERVALO_LOOP)
