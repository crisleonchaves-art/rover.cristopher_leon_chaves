# code 1.6v
import ssl
import time

import board
import socketpool
import wifi
import adafruit_hcsr04
import adafruit_minimqtt.adafruit_minimqtt as MQTT
from adafruit_io.adafruit_io import IO_MQTT
from ideaboard import IdeaBoard

# ---------------------------------------------------------------
# CONFIGURACION
# ---------------------------------------------------------------
VELOCIDAD = 0.5
DISTANCIA_MIN = 20.0       # cm: menos que esto = obstaculo
DISTANCIA_LIBRE = 400.0    # cm: valor usado cuando el sonar no recibe eco
FALLOS_PARA_LIBRE = 3      # lecturas fallidas seguidas para asumir camino libre
INTERVALO_LOOP = 0.05
INTERVALO_LECTURA = 0.15
INTERVALO_DISTANCIA = 5.0

# Claves (keys) de los feeds en Adafruit IO. Los feeds dentro de un grupo
# llevan el prefijo "grupo.": el feed "off_on" del grupo "rover" es "rover.off-on".
# (Las claves no admiten guion bajo.)
FEED_ONOFF = "rover.off-on"
FEED_ESTADO = "rover.estado"
FEED_LINEA = "rover.line"
FEED_DISTANCIA = "rover.distance"

# ---------------------------------------------------------------
# HARDWARE
# ---------------------------------------------------------------
ib = IdeaBoard()
ib.brightness = 0.2

entrada = ib.DigitalIn(board.IO32)  # True = negro (fuera), False = blanco (dentro)
sonar = adafruit_hcsr04.HCSR04(trigger_pin=board.IO19, echo_pin=board.IO18)

BLUE = (0, 0, 255)
GREEN = (0, 255, 0)
RED = (255, 0, 0)
YELLOW = (255, 255, 0)

# ---------------------------------------------------------------
# MOVIMIENTO
# ---------------------------------------------------------------
estado_actual = None


def mover(motor1, motor2, color, nombre):
    """Aplica la orden solo si es distinta a la actual (evita spam)."""
    global estado_actual
    if nombre == estado_actual:
        return
    estado_actual = nombre
    ib.motor_1.throttle = motor1
    ib.motor_2.throttle = motor2
    ib.pixel = color
    print(nombre)


def stop():
    mover(0, 0, RED, "stop")


def forward():
    mover(VELOCIDAD, VELOCIDAD, GREEN, "forward")


def back():
    mover(-VELOCIDAD, -VELOCIDAD, YELLOW, "back")


def left():
    mover(-VELOCIDAD, VELOCIDAD, BLUE, "left")


def right():
    mover(VELOCIDAD, -VELOCIDAD, BLUE, "right")


# ---------------------------------------------------------------
# WIFI
# ---------------------------------------------------------------
try:
    from secrets import secrets
except ImportError:
    print("WiFi secrets are kept in secrets.py, please add them there!")
    raise

wifi.radio.connect(secrets["ssid"], secrets["password"])

# ---------------------------------------------------------------
# ESTADO
# ---------------------------------------------------------------
activo = False
maniobra = []
paso_fin = 0.0
giro = 1

ultima_distancia = DISTANCIA_LIBRE
lectura_valida = False
fallos_sonar = 0
ultima_lectura = 0.0
ultima_publicacion = 0.0
ultima_linea = None


def iniciar_maniobra(pasos):
    global maniobra
    maniobra = list(pasos)
    siguiente_paso(time.monotonic())


def siguiente_paso(ahora):
    global paso_fin
    if maniobra:
        accion, duracion = maniobra.pop(0)
        accion()
        paso_fin = ahora + duracion


def cancelar_maniobra():
    maniobra.clear()


# ---------------------------------------------------------------
# ADAFRUIT IO (MQTT)
# ---------------------------------------------------------------
def connected(client):
    print("Adafruit IO conectado")
    client.subscribe(FEED_ONOFF)


def disconnected(client):
    print("Desconectado de Adafruit IO")


def subscribe(client, userdata, topic, granted_qos):
    print("Suscrito:", topic)


def message(client, feed_id, payload):
    global activo

    if feed_id != FEED_ONOFF:
        return

    comando = payload.strip().upper()

    if comando == "ON" and not activo:
        activo = True
        io.publish(FEED_ESTADO, "ACTIVADO")
        print("SISTEMA ENCENDIDO")

    elif comando == "OFF":
        activo = False
        cancelar_maniobra()
        stop()
        io.publish(FEED_ESTADO, "DESACTIVADO")
        print("SISTEMA APAGADO")


pool = socketpool.SocketPool(wifi.radio)

mqtt_client = MQTT.MQTT(
    broker="io.adafruit.com",
    port=1883,
    username=secrets["aio_username"],
    password=secrets["aio_key"],
    socket_pool=pool,
    ssl_context=ssl.create_default_context(),
)

io = IO_MQTT(mqtt_client)
io.on_connect = connected
io.on_disconnect = disconnected
io.on_subscribe = subscribe
io.on_message = message

io.connect()


def reconectar():
    global activo
    activo = False
    cancelar_maniobra()
    stop()

    try:
        if not wifi.radio.connected:
            wifi.radio.connect(secrets["ssid"], secrets["password"])
        io.reconnect()
        print("Reconectado")
    except Exception as e:
        print("Error:", e)
        time.sleep(2)


# ---------------------------------------------------------------
# SENSORES
# ---------------------------------------------------------------
def leer_distancia(ahora):
    """
    Devuelve la distancia en cm.
    - Lectura fallida aislada: conserva la ultima valida.
    - Varias fallidas seguidas (sin eco): asume camino libre.
    """
    global ultima_distancia, ultima_lectura, lectura_valida, fallos_sonar

    if ahora - ultima_lectura >= INTERVALO_LECTURA:
        ultima_lectura = ahora
        try:
            ultima_distancia = sonar.distance
            lectura_valida = True
            fallos_sonar = 0
        except RuntimeError:
            lectura_valida = False
            fallos_sonar += 1
            if fallos_sonar >= FALLOS_PARA_LIBRE:
                ultima_distancia = DISTANCIA_LIBRE

    return ultima_distancia


def publicar_linea(valor):
    global ultima_linea
    if valor != ultima_linea:
        io.publish(FEED_LINEA, valor)
        ultima_linea = valor


def publicar_distancia(valor, ahora):
    """Solo publica lecturas reales, como maximo cada INTERVALO_DISTANCIA s."""
    global ultima_publicacion
    if lectura_valida and ahora - ultima_publicacion >= INTERVALO_DISTANCIA:
        io.publish(FEED_DISTANCIA, valor)
        ultima_publicacion = ahora


# ---------------------------------------------------------------
# BUCLE PRINCIPAL
# ---------------------------------------------------------------
stop()

while True:
    ahora = time.monotonic()

    try:
        io.loop()
    except (MQTT.MMQTTException, OSError, RuntimeError) as e:
        print("Error de conexion:", e)
        reconectar()
        continue

    if not activo:
        time.sleep(INTERVALO_LOOP)
        continue

    try:
        # 1) Esperar a que termine el paso actual de la maniobra
        if ahora < paso_fin:
            time.sleep(INTERVALO_LOOP)
            continue

        # 2) Pasar al siguiente paso si queda alguno
        if maniobra:
            siguiente_paso(ahora)
            time.sleep(INTERVALO_LOOP)
            continue

        # 3) Decidir segun los sensores
        negro = entrada.value
        distancia = leer_distancia(ahora)

        if negro:
            publicar_linea("negro")
            giro *= -1
            iniciar_maniobra([
                (back, 0.45),
                (right if giro > 0 else left, 0.65),
            ])

        elif distancia < DISTANCIA_MIN:
            publicar_linea("blanco")
            giro *= -1
            iniciar_maniobra([
                (back, 0.4),
                (right if giro > 0 else left, 0.8),
            ])

        else:
            publicar_linea("blanco")
            forward()

        publicar_distancia(distancia, ahora)

    except (MQTT.MMQTTException, OSError) as e:
        print("Error:", e)
        reconectar()
        continue

    time.sleep(INTERVALO_LOOP)
